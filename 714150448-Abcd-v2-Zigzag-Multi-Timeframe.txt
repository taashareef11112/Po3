//@version=4
strategy("ZigZag Multi Time Frame with Filters", shorttitle="ZZMTF", overlay=true,
max_bars_back=1000)

// Input parameters
tf = input(defval='D', title="ZigZag Resolution", type=input.resolution)
prd = input(defval=2, title="ZigZag Period", minval=2, maxval=100)
showzigzag = input(defval=true, title="Show Zig Zag")
showfibo = input(defval=true, title="Show Fibonacci Ratios")
colorfulfibo = input(defval=false, title="Colorful Fibonacci Levels")
labelcol = input(defval=color.blue, title="Text Color")
fibolinecol = input(defval=color.lime, title="Line Color")
upcol = input(defval=color.lime, title="Zigzag Line Colors", inline="zzcol")
dncol = input(defval=color.red, title="", inline="zzcol")
labelloc = input(defval="Left", title="Label Location", options=["Left", "Right"])
enable236 = input(defval=true, title="Enable Level 0.236")
enable382 = input(defval=true, title="Enable Level 0.382")
enable500 = input(defval=true, title="Enable Level 0.500")
enable618 = input(defval=true, title="Enable Level 0.618")
enable786 = input(defval=true, title="Enable Level 0.786")

// Stochastic Filter
stochLength = input(14, title="Stochastic Length")
oversoldLevel = input(3, title="Oversold Level")
overboughtLevel = input(97, title="Overbought Level")
stochK = stoch(close, high, low, stochLength)
stochD = sma(stochK, 3)

// SMA Filter
smaLength = input(20, title="SMA Length")
smaClose = sma(close, smaLength)

// Regression Channel Filter
regressionLength = input(20, title="Regression Channel Length")
regressionChannel = linreg(close, regressionLength, 0)
filterRegressionChannel = close < regressionChannel

// Support/Resistance Filter with Higher Time Frame
supportResistanceLength = input(200, title="Support/Resistance Length")
higherTF = input("D", title="Higher Time Frame for Support/Resistance",
type=input.resolution)

// Calculate support and resistance levels from higher time frame
support = security(syminfo.tickerid, higherTF, low[2], lookahead=true)
resistance = security(syminfo.tickerid, higherTF, high[2], lookahead=true)

// Filter condition: close below support and close above resistance
filterSupport = close < support and close > resistance
// Plotting for visualization (optional)
plotshape(series=filterSupport, title="Filter Support/Resistance", color=color.red,
style=shape.triangledown, location=location.abovebar)


// Volatility Filter
volatilityLength = input(14, title="Volatility Length")
volatility = atr(volatilityLength)

// Bollinger Bands Filter
bbLength = input(20, title="Bollinger Bands Length")
bbMultiplier = input(2, title="Bollinger Bands Multiplier")

basis = sma(close, bbLength)
upperBand = basis + bbMultiplier * stdev(close, bbLength)
lowerBand = basis - bbMultiplier * stdev(close, bbLength)

higherTF_basis = security(syminfo.tickerid, tf, sma(close, bbLength))
higherTF_upperBand = security(syminfo.tickerid, tf, higherTF_basis + bbMultiplier *
stdev(close, bbLength))
higherTF_lowerBand = security(syminfo.tickerid, tf, higherTF_basis - bbMultiplier *
stdev(close, bbLength))

filterBB = close > lowerBand and close < upperBand and close > higherTF_lowerBand and
close < higherTF_upperBand


// Ichimoku Cloud Filter
cloudConversionLine = input(9, title="Ichimoku Conversion Line")
cloudBaseLine = input(26, title="Ichimoku Base Line")
cloudLeadingSpanB = input(52, title="Ichimoku Leading Span B")
cloudLeadingSpanDisplacement = input(26, title="Ichimoku Leading Span Displacement")

conversionLine = plot(cloudConversionLine, color=color.blue, title="Ichimoku
Conversion Line")
baseLine = plot(cloudBaseLine, color=color.red, title="Ichimoku Base Line")

bgcolor(color=color.blue, transp=90)

conversionLineValue = security(syminfo.tickerid, tf, close[cloudConversionLine])
baseLineValue = security(syminfo.tickerid, tf, close[cloudBaseLine])

crossAboveCloud = crossover(close, conversionLineValue) and crossover(close,
baseLineValue)
crossBelowCloud = crossunder(close, conversionLineValue) and crossunder(close,
baseLineValue)
// Fibonacci Retracement Filter
fibonacciFilterLength = input(20, title="Fibonacci Retracement Filter Length")
highestFibonacci = security(syminfo.tickerid, tf, highest(close,
fibonacciFilterLength))
lowestFibonacci = security(syminfo.tickerid, tf, lowest(close,
fibonacciFilterLength))
fibonacciFilter = close > lowestFibonacci and close < highestFibonacci


// ZigZag Calculation
bool newbar = change(time(tf)) != 0
bi = valuewhen(newbar, bar_index, prd - 1)
len = bar_index - bi + 1
float ph = na
float pl = na
ph := highestbars(high, nz(len, 1)) == 0 ? high : na
pl := lowestbars(low, nz(len, 1)) == 0 ? low : na

// Find Strongest and Most Potential Reversal ZigZag Points
var float strongestHigh = na
var float strongestLow = na
var float maxHigh = na
var float minLow = na

if (ph)
    maxHigh := na(maxHigh) or ph > maxHigh ? ph : maxHigh
    minLow := na(minLow) or pl < minLow ? pl : minLow
    strongestHigh := ph > maxHigh and na(strongestHigh) ? ph : strongestHigh
    strongestLow := pl < minLow and na(strongestLow) ? pl : strongestLow


// Trend Direction Filter
trendDirection = sma(close, 50) > sma(close, 200) ? 1 : sma(close, 50) < sma(close,
200) ? -1 : 0
filterTrendDirection = trendDirection == 1 ? close > sma(close, 200) : trendDirection
== -1 ? close < sma(close, 200) : false

// Filter Conditions
filterCondition = crossover(stochK, oversoldLevel) and crossover(close, smaClose) and
filterRegressionChannel and filterSupport and volatility > 0.5 * atr(14) and filterBB
and filterTrendDirection and crossover(close, cloudConversionLine) and
crossover(close, cloudBaseLine) and crossover(close, cloudLeadingSpanB) and
crossAboveCloud and fibonacciFilter


// Algoritma Harmonik ABCD
abcd_pattern_len = input(5, title="ABCD Pattern Length", minval=3, maxval=20)
isABCDValid(A, B, C, D) =>
    AB = B - A
    BC = C - B
    CD = D - C
    ABvalid = AB >= 0.618 * BC and AB <= 0.786 * BC
    BCvalid = BC >= 0.382 * AB and BC <= 0.886 * AB
    CDvalid = CD >= 1.27 * BC and CD <= 1.618 * BC

    ABvalid and BCvalid and CDvalid

plotABCDPattern() =>
    var float A = na
    var float B = na
    var float C = na
    var float D = na
    var bool validPattern = na
    if (bar_index >= abcd_pattern_len)
        A := high[abcd_pattern_len]
        B := low[abcd_pattern_len]
        C := high[1]
        D := low[0]
        validPattern := isABCDValid(A, B, C, D)
        if (validPattern)
            label.new(bar_index, na, text="ABCD", color=color.red,
style=label.style_labeldown, yloc=yloc.abovebar)




// Filter untuk Harmonic Pattern
filterHarmonicPattern = input(true, title="Filter Harmonic Pattern")
harmonicPatternLength = input(200, title="Harmonic Pattern Length", minval=1,
maxval=1000)

// Menghitung dan menyaring pola harmonik ABCD
filterABCD = security(syminfo.tickerid, higherTF, plotABCDPattern(), lookahead=true)
filterABCDLength = input(2, title="ABCD Pattern Filter Length", minval=1, maxval=100)

// Menambahkan pola harmonik ABCD ke dalam ZigZag
var bool abcdPatternDetected = na
var bool abcdCondition = na
var bool closeCondition = na

abcdCondition := na(filterABCD) ? na : true
closeCondition := na(close[abcd_pattern_len]) ? na : close >
security(syminfo.tickerid, tf, close[abcd_pattern_len])


abcdPatternDetected := abcdCondition and closeCondition


// Menyaring ZigZag berdasarkan pola harmonik ABCD
filterZigZag = na(filterABCD) ? na : true
abcdCondition := filterZigZag
closeCondition := na(close[abcd_pattern_len]) ? na : close >
security(syminfo.tickerid, tf, close[abcd_pattern_len])


plotshape(series=filterZigZag, title="Filter ZigZag by ABCD", color=color.purple,
style=shape.triangledown, location=location.abovebar)




// ZigZag Visualization
var dir = 0
dir := iff(ph and na(pl), 1, iff(pl and na(ph), -1, dir))
var max_array_size = 50
var zigzag = array.new_float(0)
oldzigzag = array.copy(zigzag)

add_to_zigzag(value, bindex) =>
    array.unshift(zigzag, bindex)
    array.unshift(zigzag, value)
    if array.size(zigzag) > max_array_size
        array.pop(zigzag)
        array.pop(zigzag)

update_zigzag(value, bindex) =>
    if array.size(zigzag) == 0
         add_to_zigzag(value, bindex)
    else
         if (dir == 1 and value > array.get(zigzag, 0)) or (dir == -1 and value <
array.get(zigzag, 0))
             array.set(zigzag, 0, value)
             array.set(zigzag, 1, bindex)
         0.

bool dirchanged = (dir != dir[1])
if ph or pl
    if dirchanged
           add_to_zigzag(dir == 1 ? ph : pl, bar_index)
    else
           update_zigzag(dir == 1 ? ph : pl, bar_index)

if showzigzag and array.size(zigzag) >= 4 and array.size(oldzigzag) >= 4
    var line zzline = na
    if array.get(zigzag, 0) != array.get(oldzigzag, 0) or array.get(zigzag, 1) !=
array.get(oldzigzag, 1)
        if array.get(zigzag, 2) == array.get(oldzigzag, 2) and array.get(zigzag, 3)
== round(array.get(oldzigzag, 3))
            line.delete(zzline)
        zzline := line.new(x1 = round(array.get(zigzag, 1)), y1 = array.get(zigzag,
0), x2 = round(array.get(zigzag, 3)), y2 = array.get(zigzag, 2), color = dir == 1 ?
upcol : dncol, width = 2)

if not showzigzag and array.size(zigzag) >= 6
    var line zzline = na
    line.delete(zzline)
    zzline := line.new(x1 = round(array.get(zigzag, 3)), y1 = array.get(zigzag, 2),
x2 = round(array.get(zigzag, 5)), y2 = array.get(zigzag, 4), color = dir == 1 ? upcol
: dncol, width = 2, style = line.style_dotted)

// Fibonacci Levels
var fibo_ratios = array.new_float(0)
var fibo_colors = array.new_color(10)
var shownlevels = 1
if barstate.isfirst
    array.push(fibo_ratios, 0.000)
    if enable236
        array.push(fibo_ratios, 0.236)
        shownlevels := shownlevels + 1
    if enable382
        array.push(fibo_ratios, 0.382)
        shownlevels := shownlevels + 1
    if enable500
        array.push(fibo_ratios, 0.500)
        shownlevels := shownlevels + 1
    if enable618
        array.push(fibo_ratios, 0.618)
        shownlevels := shownlevels + 1
    if enable786
        array.push(fibo_ratios, 0.786)
        shownlevels := shownlevels + 1
    for x = 1 to 5
        array.push(fibo_ratios, x)
        array.push(fibo_ratios, x + 0.272)
        array.push(fibo_ratios, x + 0.414)
        array.push(fibo_ratios, x + 0.618)
    // Set colors
    array.set(fibo_colors,   0,   color.lime)
    array.set(fibo_colors,   1,   color.silver)
    array.set(fibo_colors,   2,   color.gray)
    array.set(fibo_colors,   3,   color.red)
    array.set(fibo_colors,   4,   color.purple)
    array.set(fibo_colors,   5,   color.fuchsia)
    array.set(fibo_colors,   6,   color.olive)
    array.set(fibo_colors,   7,   color.navy)
    array.set(fibo_colors,   8,   color.teal)
    array.set(fibo_colors,   9,   color.orange)

var fibolines = array.new_line(0)
var fibolabels = array.new_label(0)
if showfibo and array.size(zigzag) >= 6 and barstate.islast
    if array.size(fibolines) > 0
        for x = 0 to array.size(fibolines) - 1
            line.delete(array.get(fibolines, x))
            label.delete(array.get(fibolabels, x))

    diff = array.get(zigzag, 4) - array.get(zigzag, 2)
    stopit = false
    for x = 0 to array.size(fibo_ratios) - 1
        if stopit and x > shownlevels
            break
        fibo_line_col = colorfulfibo ? array.get(fibo_colors, ((timenow / 1000) * x)
% 10) : fibolinecol
        array.unshift(fibolines,
                      line.new(x1 = round(array.get(zigzag, 5)),
                               y1 = array.get(zigzag, 2) + diff *
array.get(fibo_ratios, x),
                               x2 = bar_index,
                               y2 = array.get(zigzag, 2) + diff *
array.get(fibo_ratios, x),
                               color = fibo_line_col,
                               extend = extend.right))
        label_x_loc = labelloc == "Left" ? round(array.get(zigzag, 5)) - 1 :
bar_index
        txt = labelloc == "Left" ? "" : "                   "
        array.unshift(fibolabels,
                      label.new( x = label_x_loc,
                                  y = array.get(zigzag, 2) + diff *
array.get(fibo_ratios, x),
                                  text = txt + tostring(array.get(fibo_ratios, x),
'#.###') + "(" + tostring(round_to_mintick(array.get(zigzag, 2) + diff *
array.get(fibo_ratios, x))) + ")",
                                  textcolor = labelcol,
                                 style = label.style_none))
        if (dir == 1 and array.get(zigzag, 2) + diff * array.get(fibo_ratios, x) >
array.get(zigzag, 0)) or
           (dir == -1 and array.get(zigzag, 2) + diff * array.get(fibo_ratios, x) <
array.get(zigzag, 0))
            stopit := true
